  Запускается из workflow review.yml. Только читает код и diff, ничего не меняет.
# Выбирается в review.yml через OPENCODE_CONFIG_CONTENT = {"default_agent": "review"}:
# параметр agent у экшена игнорируется (anomalyco/opencode#36504, открыт на v1.18.32).
# primary, а не subagent: иначе OpenCode откатится на build — с shell и правом
# править файлы, то есть молча снимет все запреты ниже.
mode: primary
temperature: 0.1
# Белый список вместо чёрного: "*": deny закрывает всё, что не разрешено ниже,
# в том числе инструменты, которые появятся в новых версиях OpenCode.
permission:
  "*": deny
  # Читать изменённые файлы целиком, а не только строки из diff, и искать связанный код.
  read: allow
  glob: allow
  grep: allow
  # Вне репозитория — только diff, который review.yml готовит шагом до агента.
  # Правила задаются явно, а не наследуются: в opencode.json для /oc открыт весь /tmp
  # и ~/.config/gh.
  external_directory:
    "*": deny
    "/tmp/opencode/*": allow
  # Shell закрыт, но не целиком — только из-за бага бесплатного тарифа Zen
  # (anomalyco/opencode#51193): агент с полностью запрещённым shell получает
  # «OpenCode's free tier can only be used from within OpenCode» (так упал прогон
  # ревью на PR #19). "ask" не годится — в CI подтверждать некому, агент повиснет.
  # "pwd" без * — точное совпадение: одна команда, без аргументов и флагов.
  bash:
    "*": deny
    "pwd": allow
  # Остальное закрыто намеренно, но три пункта стоит назвать явно:
  # - edit: если агент изменит файл, экшен сам закоммитит правку и откроет PR
  #   (github.handler.ts, ветка dirty). Запрет в prompt'е — просьба, этот — барьер.
  # - bash: сначала был открыт на `git diff *` и `git log *`, и первое же авторевью
  #   (PR #19) нашло `git diff --no-index /dev/null <файл>` — чтение любого файла.
  #   Белый список команд не герметичен: у разрешённой команды всегда найдётся флаг.
  #   Поэтому diff готовит workflow, а в shell осталась только pwd (см. выше).
  # - skill: скиллы в .claude/skills описывают процесс разработки, ревью они
  #   не нужны и только тянут в контекст лишнее.

Ты делаешь ревью pull request для человека, который будет решать, мержить ли его.
Ты только читаешь код и diff и пишешь ответ — твой ответ целиком станет комментарием к PR.
Задание и формат ответа приходят в сообщении.
