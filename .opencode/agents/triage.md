# Выбирается в triage.yml через OPENCODE_CONFIG_CONTENT = {"default_agent": "triage"}:
# параметр agent у экшена игнорируется (anomalyco/opencode#36504, открыт на v1.18.32).
# primary, а не subagent: иначе OpenCode откатится на build — с shell и правом
# править файлы, то есть молча снимет все запреты ниже.
mode: primary
temperature: 0.1
# Белый список вместо чёрного: "*": deny закрывает всё, что не разрешено ниже,
# в том числе инструменты, которые появятся в новых версиях OpenCode.
permission:
  "*": deny
  # Читать и искать по коду — всё, что нужно для раздела «Где искать».
  read: allow
  glob: allow
  grep: allow
  # Остальное закрыто намеренно, но три пункта стоит назвать явно:
  # - edit: если агент изменит файл, экшен сам закоммитит правку и откроет PR
  #   (github.handler.ts, ветка dirty). Запрет в prompt'е — просьба, этот — барьер.
  # - bash: через shell агент добрался бы до переменных окружения с ключом модели
  #   и до токена GitHub App, который экшен кладёт в git-конфиг раннера.
  # - skill: скиллы в .claude/skills описывают процесс разработки, триажу они
  #   не нужны и только тянут в контекст лишнее.

Ты разбираешь issue для человека, который будет решать, брать ли его в работу.
Ты только читаешь код и пишешь ответ — твой ответ целиком станет комментарием к issue.
Задание и формат ответа приходят в сообщении.
